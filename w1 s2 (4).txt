class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        // Calculate total sum
        int totalSum = 0;
        for (int num : nums) {
            totalSum += num;
        }

        int prefixSum = 0;

        for (int i = 0; i < n; i++) {
            int current = nums[i];

            // Sum of differences with elements on the left
            int leftSum = current * i - prefixSum;

            // Sum of differences with elements on the right
            int rightSum = (totalSum - prefixSum - current)
                         - current * (n - i - 1);

            result[i] = leftSum + rightSum;

            // Add current element to prefix sum
            prefixSum += current;
        }

        return result;
    }
}
